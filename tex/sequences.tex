\subsection*{Catalan Numbers (OEIS A000108)}
\[
C_n = \frac{1}{n+1}\binom{2n}{n} = \frac{(2n)!}{(n+1)!n!}, \quad C_0 = 1
\]
Recurrence: 
\[
C_{n+1} = \sum_{i=0}^{n} C_i\,C_{n-i}
\].  

Generating function: 
\[
C(x) = \frac{1 - \sqrt{1-4x}}{2x}
\].

Applications (exact meanings):
\begin{itemize}
  \item Number of valid parenthesis expressions with \(n\) pairs.
  \item Number of monotonic lattice paths from \((0,0)\) to \((n,n)\) staying below the diagonal.
  \item Number of triangulations of a convex polygon with \(n+2\) vertices.
  \item Number of full binary trees with \(n+1\) leaves.
  \item Number of noncrossing partitions of an \(n\)-element set.
\end{itemize}

\subsection*{Bernoulli Numbers (OEIS A027641)}
\[
\frac{x}{e^x - 1} = \sum_{n=0}^\infty B_n \frac{x^n}{n!}, \quad B_0 = 1,\; B_1 = -\frac{1}{2},\; B_{2m+1} = 0\ (m \ge 1)
\]

Key formulas:
\[
\sum_{k=1}^n k^p = \frac{1}{p+1} \sum_{j=0}^{p} \binom{p+1}{j} B_j n^{p+1-j}
\]
\[
\zeta(2m) = (-1)^{m+1} \frac{(2\pi)^{2m}}{2(2m)!} B_{2m}
\]

Applications (precise uses):
\begin{itemize}
  \item Compute closed-form sums of powers up to degree \(p\).
  \item Evaluate Riemann zeta at even positive integers.
  \item Correct high-order error in Euler–Maclaurin summation for \(n\)-term truncation.
\end{itemize}

\subsection*{Stirling Numbers of the First Kind (signed) (OEIS A048994)}
\[
f_n(x) = x(x-1)\cdots(x-n+1) = \sum_{k=0}^n s(n,k)\,x^k
\]
Recurrence: 


$$ s(n,k) = s(n-1,k-1) - (n-1)s(n-1,k), s(0,0) = 1$$

Applications:
\begin{itemize}
  \item \(|s(n,k)|\): number of permutations of \(n\) elements with exactly \(k\) cycles.
  \item Convert falling factorial basis \(f_n(x)\) to power basis \(x^k\) for degree \(n\) polynomials.
\end{itemize}

\subsection*{Stirling Numbers of the Second Kind (OEIS A008277)}
\[
x^n = \sum_{k=0}^n S(n,k)\,f_k(x), \quad
S(n,k) = kS(n-1,k) + S(n-1,k-1), \quad S(0,0) = 1
\]
Closed form:
\[
S(n,k) = \frac{1}{k!} \sum_{j=0}^{k} (-1)^{k-j} \binom{k}{j} j^n
\]

Applications:
\begin{itemize}
  \item Number of ways to partition an \(n\)-element labeled set into \(k\) nonempty unlabeled blocks.
  \item Number of surjective functions from \(n\) elements onto \(k\) labeled elements (\(= k!S(n,k)\)).
  \item Coefficients when expanding monomials \(x^n\) into falling factorial basis \(f_k(x)\).
\end{itemize}

\subsection*{Bell Numbers (OEIS A000110)}
\[
B_n = \sum_{k=0}^n S(n,k), \quad 
\sum_{n=0}^\infty \frac{B_n}{n!} x^n = e^{e^x - 1}
\]
Dobinski’s formula:
\[
B_n = \frac{1}{e} \sum_{m=0}^{\infty} \frac{m^n}{m!}
\]

Applications:
\begin{itemize}
  \item Number of all set partitions of an \(n\)-element set.
  \item Number of equivalence relations on an \(n\)-element set.
  \item Number of ways to cluster \(n\) labeled objects into any number of groups.
\end{itemize}

\subsection*{Eulerian Numbers (OEIS A173018)}
Let \(A(n,k)\) = number of permutations of \(\{1,\dots,n\}\) with exactly \(k\) ascents.  
Recurrence:
\[
A(n,k) = (n-k)A(n-1,k-1) + (k+1)A(n-1,k), \quad A(1,0) = 1
\]

Exponential generating function:
\[
\sum_{n,k\ge0} A(n,k) \frac{t^n}{n!}x^k = \frac{1-x}{1 - x e^{t(1-x)}}
\]

Applications:
\begin{itemize}
  \item \(A(n,k)\): number of permutations of length \(n\) with exactly \(k\) descents.
  \item Coefficients of Eulerian polynomial \(A_n(x) = \sum_{k=0}^{n-1} A(n,k) x^k\).
  \item Expected number of descents in random permutation of length \(n\): \((n-1)/2\).
\end{itemize}

\subsection*{Computation notes:}
\begin{itemize}
  \item Catalan: compute via factorials mod \(p\).
  \item Stirling (1st/2nd): dynamic programming in \(O(n^2)\).
  \item Bell: sum \(S(n,k)\) over \(k\).
  \item Eulerian: DP with recurrence above.
  \item Bernoulli: Akiyama–Tanigawa algorithm for small \(n\).
\end{itemize}
