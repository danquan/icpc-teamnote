\subsection*{Basic: Binomial \& Multinomial}
\[
(1-x)^{-a}=\sum_{k=0}^\infty \binom{a+k-1}{k}x^k \ (|x|<1).
\]
\[
\binom{-a}{k}=(-1)^k\binom{a+k-1}{k},\qquad
\]
\[
\sum_{k=0}^n \binom{n}{k}=2^n,\quad
\sum_{k=0}^n k\binom{n}{k}=n\,2^{n-1},\quad
\sum_{k=0}^n k^2\binom{n}{k}=n(n+1)2^{n-2}.
\]

\subsection*{Hockey-stick (Christmas stocking)}
\[
\sum_{i=r}^{n}\binom{i}{r}=\binom{n+1}{r+1}\qquad (0\le r\le n).
\]

\subsection*{Vandermonde (binomial convolution)}
\[
\sum_{k}\binom{r}{k}\binom{s}{n-k}=\binom{r+s}{\,n\,}\qquad (n\in\mathbb{Z}).
\]
Notable corollary: \(\displaystyle \sum_{k=0}^n \binom{n}{k}^2=\binom{2n}{n}\).

\subsection*{Chu–Vandermonde (negative coefficients form)}
\[
\sum_{k=0}^n \binom{a}{k}\binom{b}{n-k}=\binom{a+b}{n}\quad(a,b\in\mathbb{R}),
\]
use \(\binom{-m}{k}=(-1)^k\binom{m+k-1}{k}\) to handle "selection with replacement".

\subsection*{Multinomial polynomial}
\[
(x_1+\cdots+x_m)^n=\sum_{k_1+\cdots+k_m=n}\binom{n}{k_1,\dots,k_m}\prod_{i=1}^m x_i^{k_i},
\]
\[
\binom{n}{k_1,\dots,k_m}=\frac{n!}{\prod_i k_i!},\qquad
\sum_{k_1+\cdots+k_m=n}\binom{n}{k_1,\dots,k_m}=m^n.
\]

\subsection*{Binomial transform}
\[
g_n=\sum_{k=0}^n \binom{n}{k}f_k
\ \Longleftrightarrow\
f_n=\sum_{k=0}^n (-1)^{n-k}\binom{n}{k}g_k.
\]
Power form: \(\displaystyle \sum_{k=0}^n (-1)^k\binom{n}{k}k^m=0\) for \(0\le m<n\).

\subsection*{Power sum via Bernoulli (Faulhaber)}
\[
\sum_{k=1}^{N} k^p=\frac{1}{p+1}\sum_{j=0}^{p}\binom{p+1}{j}B_j\,N^{p+1-j},
\]
\(B_j\) is Bernoulli.

Special cases for small $n$:
$$ 1 + 2 + 3 + \dots + n = \frac{n(n+1)}{2} \\$$ 
$$ 1^2 + 2^2 + 3^2 + \dots + n^2 = \frac{n(2n+1)(n+1)}{6} \\$$ 
$$ 1^3 + 2^3 + 3^3 + \dots + n^3 = \frac{n^2(n+1)^2}{4} \\$$ 
$$ 1^4 + 2^4 + 3^4 + \dots + n^4 = \frac{n(n+1)(2n+1)(3n^2 + 3n - 1)}{30} \\$$


\subsection*{Permutations with fixed points}
\[
\left|\bigcup_{i=1}^m A_i\right|
=\sum_{j\ge1}(-1)^{j-1}\!\!\sum_{1\le i_1<\cdots<i_j\le m}\!\!\!\!\!\!\!\!\! |A_{i_1}\cap\cdots\cap A_{i_j}|.
\]

Derangements (permutations with zero fixed points):
\[
!\,0=1,\quad !\,1=0,\quad !\,m=(m-1)\big(!\,(m-1)+!\,(m-2)\big),
\]
and the inclusion–exclusion form
\[
!\,m=m!\sum_{i=0}^{m}\frac{(-1)^i}{i!}.
\]

Count of permutations $\pi\in S_n$ with exactly $k$ fixed points:
\[
R(n,k)=\binom{n}{k}\,!\,(n-k),
\]

\subsection*{Related to Stirling}
Let \(f_n(x)=x(x-1)\cdots(x-n+1)\).
\[
f_n(x)=\sum_{k=0}^n s(n,k)x^k,\qquad
x^n=\sum_{k=0}^n S(n,k)\,f_k(x).
\]
Number of surjective maps \( [n]\twoheadrightarrow [k] \): \(\displaystyle k!\,S(n,k)=\sum_{j=0}^{k}(-1)^j\binom{k}{j}(k-j)^n\).

\subsection*{Some common alternating sums}
\[
\sum_{k=0}^n (-1)^k \binom{n}{k} = 0\ (n\ge1),\quad
\sum_{k=0}^n (-1)^k \binom{r+k}{k} = (-1)^n \binom{r}{n}.
\]

\subsection*{Central coefficient \& approximation}
\[
\binom{2n}{n}=\sum_{k=0}^n \binom{n}{k}^2,\qquad
\binom{2n}{n}\sim \frac{4^n}{\sqrt{\pi n}}.
\]
